- fusescope-collector version: '{{ .Values.fusescopeCollector.image.tag }}'
{{- end }}
{{- if include "fusescopeCollector.compatService" . }}

WARNING: fusescopeCollector.otlp.insecure is on. The receiver otlp/insecure takes OTLP on
{{ include "otelCollector.fullname" . }}:{{ .Values.otelCollector.ports.otlp.servicePort }} (gRPC) and :{{ index .Values.otelCollector.ports "otlp-http" "servicePort" }} (HTTP), the earlier collector's Service name and ports,
and on the collector's own Service ({{ .Values.fusescopeCollector.otlp.grpcPort }}, {{ .Values.fusescopeCollector.otlp.httpPort }}), without TLS or a key. Any workload that reaches
these ports can send, inject or forge telemetry. Admit only your emitters with
fusescopeCollector.otlp.networkPolicy.ingressFrom, keep an inventory of them, and follow the
migration guide to move each client to TLS and a key. Turn this off only when the
otelcol_receiver_{accepted,refused,failed}_*{receiver="otlp/insecure"} counters show no increase
through the guide's window and every client in the inventory is moved and verified (collector ADR 0013).
{{- else if not (include "fusescopeCollector.active" .) }}

WARNING: no collector in this release receives telemetry. Turn on a path with TLS and a key:
fusescopeAgent with fusescopeCollector.agentListener, or fusescopeCollector.rum. For a SigNoz
migration whose agents send OTLP without TLS, see fusescopeCollector.otlp.insecure (collector
ADR 0013).
{{- end }}
{{- if and .Values.fusescopeAgent.enabled .Values.fusescopeAgent.otlp.enabled }}

Pods' SDKs send OTLP to the agent on their own node (collector ADR 0014), over TLS 1.3 with the
bearer token in {{ .Values.fusescopeAgent.otlp.token.existingSecret }}:
  gRPC  https://{{ include "fusescopeAgent.otlpServiceName" . }}.{{ .Release.Namespace }}.svc:{{ .Values.fusescopeAgent.otlp.grpcPort }}  (OTEL_EXPORTER_OTLP_PROTOCOL=grpc)
  HTTP  https://{{ include "fusescopeAgent.otlpServiceName" . }}.{{ .Release.Namespace }}.svc:{{ .Values.fusescopeAgent.otlp.httpPort }}  (OTEL_EXPORTER_OTLP_PROTOCOL=http/protobuf)
Copy the CA bundle and the token into each application namespace; see the chart README.
{{- end }}

2. Get the application URL by running these commands:

{{- if .Values.signoz.ingress.enabled -}}
{{- range $host := .Values.signoz.ingress.hosts }}

  {{- range .paths }}
  http{{ if $.Values.signoz.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}

{{- end }}
{{- else if contains "NodePort" .Values.signoz.service.type }}

  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "signoz.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT

{{- else if contains "LoadBalancer" .Values.signoz.service.type }}

     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch the status of by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "signoz.fullname" . }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "signoz.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.signoz.service.port }}

{{- else if contains "ClusterIP" .Values.signoz.service.type }}

  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ .Chart.Name }},app.kubernetes.io/instance={{ .Release.Name }},app.kubernetes.io/component={{ .Values.signoz.name }}" -o jsonpath="{.items[0].metadata.name}")
  echo "Visit http://127.0.0.1:{{ .Values.signoz.service.port }} to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME {{ .Values.signoz.service.port }}:{{ .Values.signoz.service.port }}

{{- end }}
